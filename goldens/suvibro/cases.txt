# Segment duration may exceed tv; the header keeps the original sample count.
suvibro sweep=2 tv=0.1 dt=0.01 tseg=0,0.2 fseg=10,60 t1=0 t2=0
# Buffer boundaries: the full buffer fits without an end taper; leave one
# sample of room for the reference's end taper at index nt.
suvibro sweep=2 tv=65.5335 dt=0.001 tseg=0,65.5345 fseg=10,60 t1=0 t2=0
suvibro sweep=2 tv=65.5325 dt=0.001 tseg=0,65.5335 fseg=10,60 t1=0 t2=0.01
# A negative first boundary generates more samples than tv/dt, within capacity.
suvibro sweep=2 tv=0.1 dt=0.01 tseg=-0.1,0.1 fseg=10,60 t1=0 t2=0
# Multiple segments near capacity exercise the accumulated write offset.
suvibro sweep=2 tv=65.5335 dt=0.001 tseg=0,32.767,65.5345 fseg=10,30,60 t1=0 t2=0
# One boundary creates no segments, so even a large replacement tv is safe.
suvibro sweep=2 tv=0.1 dt=0.01 tseg=1000 fseg=10 t1=0 t2=0
