# Default output file and header-only mode, with one trace and one sample.
[exact] sunull nt=1 ntr=1 | sugetqcsv keys=tracl,ns,dt && cat qhed.csv
# Sample selection, custom output path and formats, odd trace and sample counts.
[exact] suspike nt=7 ntr=3 | sugetqcsv keys=tracl,dt fsamp=1 lsamp=5 formxy=%.3f formsv=%.4e qout=out.csv && cat out.csv
[exact] suspike nt=5 ntr=1 | sugetqcsv keys=tracl lsamp=99 && cat qhed.csv
[exact] suspike nt=5 ntr=1 | sugetqcsv keys=tracl fsamp=0 lsamp=0 verbose=1 && cat qhed.csv
# All special keys and positive and negative scalars.
