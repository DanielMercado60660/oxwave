# Sampling interval falls back to dt, or zero, when frequency d1 is zero.
suplane nt=64 ntr=2 | sufft | sushw key=d1 a=0 | suifft
suplane nt=64 ntr=1 | sufft | sushw key=d1,dt a=0,0 | suifft
