# Strip run-dependent PostScript metadata.
# sumax opens /dev/tty unconditionally; use a wrapper to supply its outpar in the non-tty harness.
mkdir bin; ln -s "$(command -v psgraph)" bin/psgraph; printf '#!/bin/sh\nexec "%s" outpar=/dev/null "$@"\n' "$(command -v sumax)" > bin/sumax; chmod +x bin/sumax; suplane nt=64 ntr=9 | CWPBIN="$PWD/bin" supsmax mode=max | sed -e '/^%%CreationDate/d' -e '/^%%Creator/d'
