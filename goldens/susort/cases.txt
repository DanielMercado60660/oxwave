# Wait for the named temporary file, then interrupt while stdin is still open.
# Check all installed handlers, failure status, and removal without timing output.
[stderr] mkdir tmpdir && mkfifo input && exec 3<>input && for sig in INT QUIT HUP TERM; do CWP_TMPDIR=tmpdir susort < input > out.su & pid=$!; for i in $(seq 1 200); do test -n "$(find tmpdir -type f -print)" && break; sleep 0.01; done; kill -s "$sig" "$pid"; wait "$pid"; rc=$?; test "$rc" -eq 1 && test -z "$(find tmpdir -type f -print)" || exit 2; done
# Interrupt after trace I/O has filled the named temporary file, with stdin
# still open. Each signal must exit silently with status 1 and remove the file.
[stderr] sunull nt=3 ntr=1025 > in.su && mkdir tmpdir && for sig in INT QUIT HUP TERM; do mkfifo input && exec 3<>input || exit 2; CWP_TMPDIR=tmpdir susort < input > out.su & pid=$!; cat in.su >&3; for i in $(seq 1 200); do test -n "$(find tmpdir -type f -size +0c -print)" && break; sleep 0.01; done; test -n "$(find tmpdir -type f -size +0c -print)" || exit 2; kill -s "$sig" "$pid"; wait "$pid"; rc=$?; exec 3>&-; test "$rc" -eq 1 && test ! -s out.su && test -z "$(find tmpdir -type f -print)" || exit 2; rm input; done
