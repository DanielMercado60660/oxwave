# Native binary integer output, defaults, and output parameter file.
[exact] printf '1.2 -2.5\n0 3.5\n' | a2i outpar=count > data; cat data count
[exact] printf '0.49\n-0.5\n1.5\n' | a2i n1=1 outpar=/dev/null
[exact] printf '1 2 3\n-1.5\t2.5 6e1\n' | a2i n1=3 outpar=/dev/null
[exact] printf '1 2 3\n\n' | a2i n1=0 outpar=count; cat count
[exact] a2i outpar=count </dev/null; cat count
[stderr] printf '1 bad\n' | a2i outpar=/dev/null
[stderr] printf '1\n' | a2i n1=2 outpar=/dev/null
[stderr] printf '\n' | a2i n1=1 outpar=/dev/null
