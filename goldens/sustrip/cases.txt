# Default C output, single sample and trace.
[f32] sunull nt=1 ntr=1 | sustrip outpar=par
# Odd sample and trace counts, Fortran record markers.
[exact] suplane nt=7 ntr=3 | sustrip outpar=par ftn=1
# Inspect saved native headers.
[exact] suplane nt=7 ntr=3 | sustrip head=headers outpar=par > samples; cat headers
# Inspect parameter file and non-default interval.
sunull nt=5 ntr=2 dt=0.004 | sustrip head=headers outpar=par > samples; cat par
[f32] suplane nt=13 ntr=1 | sustrip outpar=par ftn=0
# Packed byte and short traces.
[exact] suplane nt=64 ntr=2 | supack1 gpow=0.5 | sustrip outpar=par
[exact] suplane nt=64 ntr=2 | supack2 gpow=0.5 | sustrip outpar=par ftn=1
[stderr] sunull nt=3 ntr=1 | sustrip outpar=par verbose=1 > samples
