# Default tolerance, one sample and one trace.
suplane nt=1 ntr=1 > a && sucmp a a
# Odd trace size and explicit zero tolerance.
suplane nt=17 ntr=3 > a && sucmp a a limit=0
# Header mismatch.
suplane nt=17 ntr=3 > a && sushw < a key=cdp a=42 > b && sucmp a b
# Different trace count.
suplane nt=17 ntr=3 > a && dd if=a of=b bs=308 count=2 2>/dev/null && sucmp a b
# Different sample count.
suplane nt=17 ntr=2 > a && suplane nt=19 ntr=2 > b && sucmp a b
# Positive and negative samples, within and outside tolerance.
sunull nt=9 ntr=1 | sugain bias=1 > a && sugain < a scale=1.001 > b && sucmp a b limit=0.01
sunull nt=9 ntr=1 | sugain bias=1 > a && sugain < a scale=1.1 > b && sucmp a b limit=0.01
sunull nt=9 ntr=1 | sugain bias=-1 > a && sugain < a scale=1.1 > b && sucmp a b limit=0.01
# Zero in A is ignored; negative limit makes identical nonzero samples differ.
sunull nt=9 ntr=1 > a && sugain < a bias=1 > b && sucmp a b
sunull nt=9 ntr=1 | sugain bias=1 > a && sucmp a a limit=-1
# Diagnostics and errors.
[stderr] suplane nt=7 ntr=1 > a && sucmp a a verbose=1
[stderr] suplane nt=7 ntr=1 > a && sucmp a a unexpected=1
sucmp missing_a missing_b
touch a b && sucmp a b
sucmp only_one
sucmp a b limit=1e999
