suspike nt=7 ntr=3 | sushw key=sx,sy,gx,gy,selev,gelev,sdepth,gdel,sdel,swdep,gwdep a=123,-456,789,321,34,-45,10,20,-30,40,50 | sulcthw xt=12.5 yt=-3 zt=4

suspike nt=5 ntr=3 | sushw key=sx,sy,gx,gy,selev,gelev,scalco,scalel a=100,-200,300,400,50,-60,-100,10 b=10,20,-30,40,0,0,0,0 | sulcthw xr=10 yr=-20 zr=30

suspike nt=3 ntr=1 | sulcthw xr=90

suspike nt=3 ntr=1 | sushw key=sx,sy a=100,50 | sulcthw zr=-90
