# Native signed integers, default width, and the output parameter file.
python3 -c 'import sys,struct; sys.stdout.buffer.write(struct.pack("=4i", 0, -1, 2147483647, -2147483648))' | i2a outpar=count; cat count
# Single-element and odd-width rows.
python3 -c 'import sys,struct; sys.stdout.buffer.write(struct.pack("=3i", -123, 0, 456))' | i2a n1=1 outpar=count; cat count
python3 -c 'import sys,struct; sys.stdout.buffer.write(struct.pack("=6i", 1, 2, 3, -4, -5, -6))' | i2a n1=3 outpar=count; cat count
# Empty input and a zero width.
i2a outpar=count </dev/null; cat count
python3 -c 'import sys,struct; sys.stdout.buffer.write(struct.pack("=i", 7))' | i2a n1=0 outpar=count; cat count
